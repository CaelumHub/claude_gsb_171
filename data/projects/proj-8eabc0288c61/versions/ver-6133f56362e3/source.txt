// 列表操作与循环（演示内存模型 / 下标访问）
var data = [3, 1, 4, 1, 5, 9, 2, 6];
var sum = 0;

for (var i = 0; i < len(data); i = i + 1) {
    sum += data[i];
}
print("总和 =", sum);
print("平均值 =", sum / len(data));

push(data, 100);
print("追加后长度 =", len(data));
print("最后一个元素 =", data[len(data) - 1]);

var evens = [];
for (var j = 0; j < len(data); j = j + 1) {
    if (data[j] % 2 == 0) {
        push(evens, data[j]);
    }
}
print("偶数 =", evens);
